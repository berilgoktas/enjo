# Sunucuda PowerShell'de çalıştırılacak komutlar

# 1. Dosyanın olduğu klasöre git (dosyanızı nereye attıysanız oraya)
cd C:\Users\[KULLANICI_ADI]\Desktop

# 2. Docker image'ı yükle
docker load -i C:\Users\[KULLANICI_ADI]\Desktop\enjoapifront.tar

# 3. Image'ın yüklendiğini kontrol et
docker images

# 4. Eski container'ı durdur ve sil (eğer varsa)
docker stop enjoapifront-container
docker rm enjoapifront-container

# 5. Container'ı farklı portta çalıştır
# -d: arka planda çalıştır
# -p 3445:80: dışarıdan 3445 portu, içeriden 80 portu
# --name: container adı
# --restart unless-stopped: sunucu yeniden başladığında otomatik başlat
docker run -d -p 3445:80 --name enjoapifront-container --restart unless-stopped enjoapifront:latest

# 6. Container'ın çalıştığını kontrol et
docker ps

# Uygulama şu adresten erişilebilir: http://localhost:3445
